test('global Run emits typed default source settings and selected tracks',async ({page})=>{
  await page.getByRole('button',{name:'Add Files…'}).click();
  await page.getByRole('button',{name:'Load Tracks'}).click();
  await page.getByRole('button',{name:'Quick Convert',exact:true}).click();
  const quick=page.getByRole('region',{name:'Quick Convert',exact:true});
  await quick.getByLabel('Output Path',{exact:true}).fill('/media/output');
  await page.getByRole('button',{name:'Run: Encode',exact:true}).click();
  await expect.poll(async()=>page.evaluate(()=> (globalThis as any).__sourceCalls.some((call:any)=>call.command==='mock_start_encode'))).toBe(true);

  expect(request.source).toMatchObject({inputPath:'/media/input.mkv',outputPath:'/media/output.mkv',streamIndices:[0,1,2]});
  expect(request.settings).toMatchObject({backend:'standalone',encoder:'svtAv1',crf:30,svtCrfQuarterSteps:120,audio:[{streamIndex:1,codec:'opus',bitrateKbps:128,channels:'stereo'}]});
